"use client";
import { useEffect, useState } from "react";
import { useTranslations } from "next-intl";
import { useRouter } from "next/navigation";
import { createProductAction } from "@/app/actions/product.actions";
import { generateBarcode, blockchainHash } from "@/lib/barcode";

export default function NewProductPage() {
  const t = useTranslations("product");
  const router = useRouter();
  const [form, setForm] = useState({
    name: "", category: "", barcode: "", purchasePrice: "", salePrice: "",
    quantity: "", expiryDate: "", imageUrl: "", branchId: "",
  });
  const [hash, setHash] = useState("");
  const [branches, setBranches] = useState<{ id: string; name: string }[]>([]);
  const [saving, setSaving] = useState(false);

  useEffect(() => {
    fetch("/api/branches").then((r) => r.json()).then(setBranches);
    regen();
  }, []);

  function regen() {
    const bc = generateBarcode();
    setForm((f) => ({ ...f, barcode: bc }));
    setHash(blockchainHash({ barcode: bc, ts: Date.now() }));
  }

  async function uploadImage(file: File) {
    const reader = new FileReader();
    reader.onload = () => setForm((f) => ({ ...f, imageUrl: reader.result as string }));
    reader.readAsDataURL(file);
  }

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    setSaving(true);
    try {
      await createProductAction({
        name: form.name, category: form.category,
        purchasePrice: +form.purchasePrice, salePrice: +form.salePrice,
        quantity: +form.quantity, expiryDate: form.expiryDate,
        imageUrl: form.imageUrl || undefined, branchId: form.branchId,
      });
      router.push("/dashboard");
    } finally { setSaving(false); }
  }

  const inputCls = "w-full rounded-lg border p-3 dark:bg-gray-800 dark:border-gray-700";

  return (
    <div className="max-w-3xl mx-auto p-6">
      <h1 className="text-2xl font-bold mb-6">➕ {t("addNew")}</h1>
      <form onSubmit={submit} className="space-y-4 rounded-xl bg-white dark:bg-gray-900 shadow p-6">
        <div className="grid md:grid-cols-2 gap-4">
          <input required placeholder={t("name")} value={form.name}
            onChange={(e) => setForm({ ...form, name: e.target.value })} className={inputCls} />
          <select required value={form.category} onChange={(e) => setForm({ ...form, category: e.target.value })} className={inputCls}>
            <option value="">{t("category")}</option>
            {["أدوية", "مواد غذائية", "مشروبات", "منظفات", "إلكترونيات", "ملابس"].map((c) => <option key={c}>{c}</option>)}
          </select>
          <div className="flex gap-2">
            <input required dir="ltr" placeholder={t("barcode")} value={form.barcode} readOnly className={`${inputCls} flex-1`} />
            <button type="button" onClick={regen} className="rounded-lg bg-blue-600 text-white px-4 whitespace-nowrap">🔄 {t("generate")}</button>
          </div>
          <input dir="ltr" readOnly value={hash.slice(0, 24) + "..."} className={`{inputCls} text-xs text-gray-400`} title={`Hash:{hash}`} />
          <input required type="number" step="0.01" min="0" placeholder={t("purchasePrice")} value={form.purchasePrice}
            onChange={(e) => setForm({ ...form, purchasePrice: e.target.value })} className={inputCls} />
          <input required type="number" step="0.01" min="0" placeholder={t("salePrice")} value={form.salePrice}
            onChange={(e) => setForm({ ...form, salePrice: e.target.value })} className={inputCls} />
          <input required type="number" min="0" placeholder={t("quantity")} value={form.quantity}
            onChange={(e) => setForm({ ...form, quantity: e.target.value })} className={inputCls} />
          <input required type="date" value={form.expiryDate}
            onChange={(e) => setForm({ ...form, expiryDate: e.target.value })} className={inputCls} />
          <select required value={form.branchId} onChange={(e) => setForm({ ...form, branchId: e.target.value })} className={inputCls}>
            <option value="">{t("branch")}</option>
            {branches.map((b) => <option key={b.id} value={b.id}>{b.name}</option>)}
          </select>
          <label className={`${inputCls} cursor-pointer flex items-center justify-center text-gray-400`}>
            📷 {form.imageUrl ? "✅ " + t("image") : t("image")}
            <input type="file" accept="image/*" hidden onChange={(e) => e.target.files?.[0] && uploadImage(e.target.files[0])} />
          </label>
        </div>
        <button disabled={saving} className="w-full rounded-lg bg-blue-600 text-white p-3 font-bold hover:bg-blue-700 disabled:opacity-50">
          {saving ? "..." : `💾 ${t("save")}`}
        </button>
      </form>
    </div>
  );
}
