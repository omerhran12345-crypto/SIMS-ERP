"use client";
import { useState } from "react";
import Link from "next/link";
import { useTranslations } from "next-intl";
import ThemeToggle from "@/components/ThemeToggle";
import LangToggle from "@/components/LangToggle";
import { forgotPasswordAction, resetPasswordAction } from "@/app/actions/auth.actions";

export default function ForgotPasswordPage() {
  const t = useTranslations("forgot");
  const [step, setStep] = useState<1 | 2>(1);
  const [email, setEmail] = useState("");
  const [otp, setOtp] = useState("");
  const [newPassword, setNewPassword] = useState("");
  const [msg, setMsg] = useState("");
  const [error, setError] = useState("");

  async function sendOtp(e: React.FormEvent) {
    e.preventDefault();
    setError(""); setMsg("");
    try {
      await forgotPasswordAction(email);
      setMsg(t("otpSent"));
      setStep(2);
    } catch { setError(t("notFound")); }
  }

  async function reset(e: React.FormEvent) {
    e.preventDefault();
    setError("");
    try {
      await resetPasswordAction(email, otp, newPassword);
      setMsg(t("resetDone"));
    } catch { setError(t("invalidOtp")); }
  }

  const inputCls = "w-full rounded-lg border p-3 dark:bg-gray-800 dark:border-gray-700";

  return (
    <main className="min-h-screen flex items-center justify-center bg-gradient-to-br from-blue-50 to-indigo-100 dark:from-gray-950 dark:to-gray-900 p-4">
      <ThemeToggle /><LangToggle />
      <form onSubmit={step === 1 ? sendOtp : reset}
        className="w-full max-w-md bg-white dark:bg-gray-900 rounded-2xl shadow-xl p-8 space-y-4">
        <h1 className="text-2xl font-bold text-center">🔑 {t("title")}</h1>

        {error && <p className="bg-red-100 text-red-700 rounded-lg p-3 text-sm text-center">{error}</p>}
        {msg && <p className="bg-green-100 text-green-700 rounded-lg p-3 text-sm text-center">{msg}</p>}

        {step === 1 ? (
          <>
            <input type="email" required dir="ltr" placeholder={t("email")} value={email}
              onChange={(e) => setEmail(e.target.value)} className={inputCls} />
            <button className="w-full rounded-lg bg-blue-600 text-white p-3 font-bold hover:bg-blue-700">
              📨 {t("sendOtp")}
            </button>
          </>
        ) : (
          <>
            <input required dir="ltr" maxLength={6} placeholder={t("otp")} value={otp}
              onChange={(e) => setOtp(e.target.value)} className={`${inputCls} text-center tracking-widest`} />
            <input type="password" required minLength={6} placeholder={t("newPassword")} value={newPassword}
              onChange={(e) => setNewPassword(e.target.value)} className={inputCls} />
            <button className="w-full rounded-lg bg-green-600 text-white p-3 font-bold hover:bg-green-700">
              ✅ {t("resetBtn")}
            </button>
          </>
        )}

        <Link href="/login" className="block text-center text-sm text-blue-600 hover:underline">
          ← {t("backToLogin")}
        </Link>
      </form>
    </main>
  );
}
