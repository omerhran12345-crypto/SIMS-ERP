"use client";
import { useEffect, useState } from "react";
import { useTranslations } from "next-intl";
import DataTable, { Column } from "@/components/ui/DataTable";
import ExportExcel from "@/components/ui/ExportExcel";
import PrintButton from "@/components/PrintButton";

type Tab = "COA" | "JE" | "TB" | "PL" | "BS";
const TABS: Tab[] = ["COA", "JE", "TB", "PL", "BS"];

interface AnyRow { [k: string]: any; }

export default function AccountingPage() {
  const t = useTranslations("accounting");
  const [tab, setTab] = useState<Tab>("COA");
  const [rows, setRows] = useState<AnyRow[]>([]);

  useEffect(() => {
    fetch(`/api/accounting?report=${tab}`).then((r) => r.json()).then(setRows);
  }, [tab]);

  const cols: Record<Tab, Column<AnyRow>[]> = {
    COA: [
      { key: "code", label: t("code") }, { key: "name", label: t("name") },
      { key: "type", label: t("type") },
      { key: "balance", label: t("balance"), render: (r) => r.balance.toFixed(2) },
    ],
    JE: [
      { key: "date", label: t("date"), render: (r) => new Date(r.date).toLocaleDateString() },
      { key: "reference", label: t("ref") },
      { key: "description", label: t("desc") },
      { key: "debitAccount", label: t("debit") },
      { key: "creditAccount", label: t("credit") },
      { key: "amount", label: t("amount"), render: (r) => r.amount.toFixed(2) },
    ],
    TB: [
      { key: "code", label: t("code") }, { key: "name", label: t("name") },
      { key: "debit", label: t("debit"), render: (r) => r.debit.toFixed(2) },
      { key: "credit", label: t("credit"), render: (r) => r.credit.toFixed(2) },
    ],
    PL: [
      { key: "line", label: t("line") },
      { key: "amount", label: t("amount"), render: (r) => r.amount.toFixed(2) },
    ],
    BS: [
      { key: "line", label: t("line") },
      { key: "amount", label: t("amount"), render: (r) => r.amount.toFixed(2) },
    ],
  };

  return (
    <div className="space-y-4 p-6">
      <h1 className="text-2xl font-bold">📒 {t("title")}</h1>
      <div className="flex flex-wrap gap-2">
        {TABS.map((x) => (
          <button key={x} onClick={() => setTab(x)}
            className={`rounded-lg px-4 py-2 text-sm font-bold ${tab === x ? "bg-blue-600 text-white" : "bg-gray-100 dark:bg-gray-800"}`}>
            {t(x.toLowerCase())}
          </button>
        ))}
      </div>
      <DataTable columns={cols[tab]} rows={rows} filename={`report_${tab.toLowerCase()}`} />
      <div className="flex gap-2 justify-end">
        <ExportExcel data={rows} filename={`report_${tab.toLowerCase()}`} /><PrintButton />
      </div>
    </div>
  );
}
