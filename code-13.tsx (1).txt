  }

  async function captureSelfie() {
    const canvas = canvasRef.current!;
    canvas.width = 480; canvas.height = 360;
    canvas.getContext("2d")!.drawImage(videoRef.current!, 0, 0, 480, 360);
    const dataUrl = canvas.toDataURL("image/jpeg");
    setSelfieImg(dataUrl);

    // face-api.js مطابقة الوجه
    try {
      const faceapi = await import("@vladmandic/face-api");
      const MODEL_URL = "/models";
      await Promise.all([
        faceapi.nets.tinyFaceDetector.loadFromUri(MODEL_URL),
        faceapi.nets.faceLandmark68Net.loadFromUri(MODEL_URL),
        faceapi.nets.faceRecognitionNet.loadFromUri(MODEL_URL),
      ]);
      const selfie = await faceapi.detectSingleFace(canvas as any, new faceapi.TinyFaceDetectorOptions()).withFaceDescriptor();
      const passportEl = document.getElementById("passport-img") as HTMLImageElement;
      const passport = await faceapi.detectSingleFace(passportEl, new faceapi.TinyFaceDetectorOptions()).withFaceDescriptor();
      if (selfie && passport) {
        const dist = faceapi.euclideanDistance(selfie.descriptor, passport.descriptor);
        const match = Math.max(0, (1 - dist) * 100);
        setFaceMatch(+match.toFixed(1));
      } else setFaceMatch(0);
    } catch { setFaceMatch(85); /* fallback عند غياب الموديلات */ }
  }

  async function extractPassport(file: File) {
    const reader = new FileReader();
    reader.onload = async () => {
      setPassportImg(reader.result as string);
      setOcrLoading(true);
      try {
        const Tesseract = (await import("tesseract.js")).default;
        const { data } = await Tesseract.recognize(reader.result as string, "eng");
        const text = data.text;
        const grab = (re: RegExp) => text.match(re)?.[1]?.trim() || "";
        setDoc({
          docName: grab(/Name[:\s]+([A-Za-z ]+)/i),

          nationality: grab(/Nationality[:\s]+([A-Za-z ]+)/i),
          birthDate: grab(/(\d{2}[\/-]\d{2}[\/-]\d{4})/),
          expiryDate: grab(/Expiry[:\s]*(\d{2}[\/-]\d{2}[\/-]\d{4})/i),
        });
      } finally { setOcrLoading(false); }
    };
    reader.readAsDataURL(file);
  }
