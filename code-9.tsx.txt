"use client";
import { useEffect, useState } from "react";
import { useTranslations } from "next-intl";
import DataTable, { Column } from "@/components/ui/DataTable";
import ExportExcel from "@/components/ui/ExportExcel";
import PrintButton from "@/components/PrintButton";

interface Row { id: string; name: string; quantity: number; expiryDate: string; }
interface Check { productId: string; status: string; notes: string; imageUrl?: string; }

export default function QualityPage() {
  const t = useTranslations("quality");
  const [rows, setRows] = useState<Row[]>([]);
  const [open, setOpen] = useState(false);
  const [form, setForm] = useState<Check>({ productId: "", status: "PASS", notes: "" });
  const inputCls = "w-full rounded-lg border p-2 dark:bg-gray-800 dark:border-gray-700";

  useEffect(() => {
    fetch("/api/quality/expiring").then((r) => r.json()).then(setRows);
  }, []);

  async function saveCheck() {
    await fetch("/api/quality/checks", {
      method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(form),
    });
    setOpen(false); setForm({ productId: "", status: "PASS", notes: "" });
  }

  const cols: Column<Row>[] = [
    { key: "name", label: t("product") },
    { key: "quantity", label: t("qty") },
    { key: "expiryDate", label: t("expiry"), render: (r) => (
      <span className="text-red-600 font-semibold">{new Date(r.expiryDate).toLocaleDateString()}</span>
    )},
  ];

  return (
    <div className="space-y-4 p-6">
      <div className="flex items-center justify-between">
        <h1 className="text-2xl font-bold">🔬 {t("title")}</h1>
        <button onClick={() => setOpen(true)}
          className="rounded-lg bg-blue-600 text-white px-5 py-2 font-bold hover:bg-blue-700">➕ {t("newCheck")}</button>
      </div>

      <DataTable columns={cols} rows={rows} filename="expiring_products" />

      <div className="flex gap-2 justify-end">
        <ExportExcel data={rows} filename="expiring_products" /><PrintButton />
      </div>

      {open && (
        <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/60 p-4" onClick={() => setOpen(false)}>
          <div className="w-full max-w-md bg-white dark:bg-gray-900 rounded-xl p-5 space-y-3" onClick={(e) => e.stopPropagation()}>
            <h2 className="font-bold text-lg">🔬 {t("newCheck")}</h2>
            <select required value={form.productId} onChange={(e) => setForm({ ...form, productId: e.target.value })} className={inputCls}>
              <option value="">{t("selectProduct")}</option>
              {rows.map((r) => <option key={r.id} value={r.id}>{r.name}</option>)}
            </select>
            <select value={form.status} onChange={(e) => setForm({ ...form, status: e.target.value })} className={inputCls}>
              <option value="PASS">✅ {t("pass")}</option>
              <option value="FAIL">❌ {t("fail")}</option>
              <option value="REWORK">♻️ {t("rework")}</option>
            </select>
            <textarea placeholder={t("notes")} value={form.notes}
              onChange={(e) => setForm({ ...form, notes: e.target.value })} className={inputCls} rows={3} />
            <label className={`${inputCls} cursor-pointer flex items-center justify-center text-gray-400`}>
              📷 {t("uploadPhoto")}
              <input type="file" accept="image/*" hidden onChange={(e) => {
                const f = e.target.files?.[0]; if (!f) return;
                const r = new FileReader(); r.onload = () => setForm((x) => ({ ...x, imageUrl: r.result as string })); r.readAsDataURL(f);
              }} />
            </label>
            {form.imageUrl && <img src={form.imageUrl} alt="" className="h-24 rounded object-cover" />}
            <button onClick={saveCheck} className="w-full rounded-lg bg-green-600 text-white p-3 font-bold">💾 {t("save")}</button>
          </div>
        </div>
      )}
    </div>
  );
}
