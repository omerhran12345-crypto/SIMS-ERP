"use client";
import { useRouter } from "next/navigation";

export default function LangToggle() {
  const router = useRouter();

  const toggle = () => {
    const cur = localStorage.getItem("sims_lang") || "ar";
    const next = cur === "ar" ? "en" : "ar";
    localStorage.setItem("sims_lang", next);
    document.documentElement.lang = next;
    document.documentElement.dir = next === "ar" ? "rtl" : "ltr";
    router.refresh();
  };

  return (
    <button onClick={toggle}
      className="fixed top-4 right-4 z-50 rounded-full bg-blue-600 text-white px-4 py-2 text-sm font-bold shadow">
      🌐 العربية / English
    </button>
  );
}
